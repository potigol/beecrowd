## 2104 - Lasers

Descrição do Problema: [2104 - Lasers](https://resources.beecrowd.com/repository/UOJ_2104.html)

Categoria: [Geometria Computacional](https://potigol.github.io/beecrowd/categorias/geometriacomputacional/)

```poti title="2104 - Lasers" linenums="1"
tipo Ponto = (Real, Real)
tipo Pontos = Lista[Ponto]

def cross(a, b, c: Ponto) =
  (b.primeiro - a.primeiro) * (c.segundo - a.segundo) -
  (b.segundo - a.segundo) * (c.primeiro - a.primeiro)

def cons(pontos: Pontos, ponto: Ponto): Pontos =
  se pontos.tamanho < 2 ou cross(pontos[2], pontos[1], ponto) > 0
    então ponto :: pontos
    senão cons(pontos.cauda, ponto)
  fim

val vazia: Pontos = []

def constroi(pontos: Pontos) = pontos.injete(vazia)(cons).inverta

def fecho(pontos: Pontos)
  p = pontos.ordene
  baixo = constroi(p)
  cima = constroi(p.inverta)
  retorne (baixo.cauda + cima.cauda).inverta
fim

def area(pontos: Pontos) = soma(pontos, pontos.cabeça, 0.0).abs / 2

def termo(p, q: Ponto) = p.primeiro * q.segundo - q.primeiro * p.segundo

def soma(pontos: Pontos, primeiro: Ponto, acumulado: Real): Real =
  escolha pontos
    caso []      => acumulado
    caso [p]     => acumulado + termo(p, primeiro)
    caso p :: ps => soma(ps, primeiro, acumulado + termo(p, ps.cabeça))
  fim

def projeção(xp, yp, zp: Real) = (x, y, z: Real) =>
   (xp + (x - xp) * zp / (zp - z), yp + (y - yp) * zp / (zp - z))
  
var  n := leia_inteiro 
enquanto n > 0 faça
  a = leia_reais(" ")
  proj = projeção(a[1], a[2], a[3])
  pontos = para i de 1 até n gere
    b = leia_reais(" ")
    proj(b[1], b[2], b[3])
  fim

  escreva area(fecho(pontos)) formato "%.2f"
  n := leia_inteiro 
fim
```
