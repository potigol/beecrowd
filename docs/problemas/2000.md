## 2018 - Olimpíadas de Natal

Descrição do Problema: [2018 - Olimpíadas de Natal](https://resources.beecrowd.com/repository/UOJ_2018.html)

```poti title="2018 - Olimpíadas de Natal" linenums="1"
tipo Pais
    nome: Texto
    ouro, prata, bronze: Inteiro
    pontos = ouro * 1000000 + prata * 1000 + bronze
    texto() = "{nome} {ouro} {prata} {bronze}"
fim

soma(a: Lista[Pais]): Pais
  pontos = a.injete(0)(_ + _.pontos)
  ouro = pontos div 1000000
  prata = (pontos div 1000) mod 1000
  bronze = pontos mod 1000
  Pais(a[1].nome, ouro, prata, bronze)
fim


var paises: Lista[Pais] := []
leia_texto
enquanto não eof faça
  a, b, c = leia_texto
  paises := Pais(a, 1, 0, 0) :: Pais(b, 0, 1, 0) :: Pais(c, 0, 0, 1) :: paises
  leia_texto
fim

ordem = paises.ordene(_.nome)
            .divida_quando(_.nome <> _.nome)
            .mapeie(soma)
            .ordene(_.nome)
            .ordene(- _.pontos)

escreva "Quadro de Medalhas"
escreva ordem.mapeie(_.texto).junte("\n")
```

## 2087 - Conjuntos Bons e Ruins

```poti title="2087 - Conjuntos Bons e Ruins" linenums="1"
var n := leia_inteiro
enquanto n > 0 faça
  palavras = leia_textos(n).ordene.mutável
  var bom := verdadeiro
  var i := 1
  enquanto i < n e bom faça
    se palavras[i].tamanho <= palavras[i+1].tamanho e
       palavras[i] == palavras[i+1].pegue(palavras[i].tamanho) então
      bom := falso
    fim
    i := i + 1
  fim
  se bom
    então escreva "Conjunto Bom"
    senão escreva "Conjunto Ruim"
  fim
  n := leia_inteiro
fim
```
