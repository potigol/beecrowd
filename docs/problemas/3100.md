```poti title="3136 - Arquitetura de Pilha" linenums="1"
reg(comando: Lista[Texto]) = comando[2].cauda.inteiro + 1
```

Categoria: [Iniciante](https://potigol.github.io/beecrowd/categorias/iniciante/)

```poti title="3162 - Comunicação Espacial" linenums="1"
n = leia_inteiro
ss = para i de 1 até n gere leia_inteiros(" ") fim
para i de 1 até n faça
  var distancia := 500.0
  para j de 1 até n se i <> j faça
    s, r = ss[i], ss[j]
    a, b, c = s[1] - r[1], s[2] - r[2], s[3] - r[3]
    dif = raiz(a^2 + b^2 + c^2)
    se dif < distancia então distancia := dif fim
  fim
  se distancia <= 20 então
    escreva "A"
  senãose distancia <= 50 então
    escreva "M"
  senão
    escreva "B"
  fim
fim
```

```poti title="3166 - Procurando Palavras na Diagonal Principal" linenums="1"
contém(texto, palavra: Texto) =
    texto.contains(palavra) ou texto.contains(palavra.inverta)

entrada = leia_inteiros(" ")
```
